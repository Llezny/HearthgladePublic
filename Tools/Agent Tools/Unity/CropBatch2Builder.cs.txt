    public static void Build() {
        try {
            ImportIcons();
            ImportModels();
            AssetDatabase.Refresh();
            foreach( var spec in Specs ) {
                BuildOne( spec );
            }
            AssetDatabase.SaveAssets();
            global::Editor.EditorScripts.RefreshItemsDatabase();
            AssetDatabase.SaveAssets();
            Debug.Log( "CropBatch2Builder: done" );
        } catch( Exception e ) {
            Debug.LogError( "CropBatch2Builder failed: " + e );
            EditorApplication.Exit( 1 );
        }
    }

    // ---- import ---------------------------------------------------------------------------------------------------

    static void ImportIcons() {
        var template = ( TextureImporter ) AssetImporter.GetAtPath( IconResources + "StrawberrySeed.png" );
        var settings = new TextureImporterSettings();
        template.ReadTextureSettings( settings );
        var paths = new List<string>();
        foreach( var spec in Specs ) {
            if( spec.ProduceIcon != null ) {
                paths.Add( spec.ProduceIcon );
            }
            paths.Add( spec.SeedIcon );
        }
        foreach( var path in paths.Distinct() ) {
            AssetDatabase.ImportAsset( path, ImportAssetOptions.ForceSynchronousImport );
            var importer = ( TextureImporter ) AssetImporter.GetAtPath( path );
            importer.textureType = TextureImporterType.Sprite;
            importer.SetTextureSettings( settings );
            importer.textureCompression = template.textureCompression;
            importer.compressionQuality = template.compressionQuality;
            importer.maxTextureSize = template.maxTextureSize;
            importer.SaveAndReimport();
        }
    }

    static void ImportModels() {
        foreach( var spec in Specs ) {
            string path = ModelDir + spec.Name + ".fbx";
            AssetDatabase.ImportAsset( path, ImportAssetOptions.ForceSynchronousImport );
            var importer = ( ModelImporter ) AssetImporter.GetAtPath( path );
            importer.materialImportMode = ModelImporterMaterialImportMode.None;
            importer.isReadable = false;
            importer.importCameras = false;
            importer.importLights = false;
            importer.importBlendShapes = false;
            importer.generateSecondaryUV = false;
            importer.meshCompression = ModelImporterMeshCompression.Off;
            importer.animationType = ModelImporterAnimationType.None;
            importer.importAnimation = false;
            importer.SaveAndReimport();
        }
    }

    // ---- one crop -------------------------------------------------------------------------------------------------

    static void BuildOne( Spec spec ) {
        var meshes = LoadMeshes( spec.Name );
        var material = AssetDatabase.LoadAssetAtPath<Material>( MaterialPath );
        string produceAsset = spec.ProduceAsset ?? spec.Name;

        Hearthglade.Gameplay.Items.ItemSO produce;
        if( spec.ExistingProduce != null ) {
            produce = AssetDatabase.LoadAssetAtPath<Hearthglade.Gameplay.Items.ItemSO>( spec.ExistingProduce );
        } else {
            produce = CopyScriptable<Hearthglade.Gameplay.Items.ItemSO>( ItemsDir + spec.ProduceFolder + produceAsset + ".asset",
                ItemsDir + spec.ProduceTemplate + ".asset", so => {
                    Set( so, "itemName", spec.Display );
                    Set( so, "itemDescription", spec.Description );
                    Set( so, "maxItemsInStack", spec.Stack );
                    Set( so, "ItemRarity", 0, optional: true );
                    SetIcon( so, spec.ProduceIcon );
                    // fibres (Flax, Cotton) and plain ingredients (grain) have no healing fields
                    Set( so, "foodType", ( int ) spec.Food, optional: true );
                    Set( so, "NutritionQuality", 1f, optional: true );
                    Set( so, "thirstHealing", spec.Thirst, optional: true );
                    Set( so, "hungerHealing", spec.Hunger, optional: true );
                    Set( so, "healthHealing", spec.Health, optional: true );
                } );
        }
        var seed = CopyScriptable<Hearthglade.Gameplay.Items.ItemSO>( ItemsDir + "Resources/" + spec.SeedAsset + ".asset",
            ItemsDir + ( spec.SeedTemplate ?? "Resources/StrawberrySeed" ) + ".asset", so => {
                Set( so, "itemName", spec.SeedName );
                Set( so, "itemDescription", spec.SeedDescription );
                SetIcon( so, spec.SeedIcon );
            } );

        GameObject wildPrefab;
        if( spec.ExistingWildPrefab != null ) {
            // the wild watermelon already exists: it just starts dropping the seed
            var existing = AssetDatabase.LoadAssetAtPath<ResourceSO>( spec.ExistingWildResource );
            var eso = new SerializedObject( existing );
            Set( eso, "BonusItem", seed );
            Set( eso, "BonusChance", spec.WildSeedChance );
            eso.ApplyModifiedPropertiesWithoutUndo();
            EditorUtility.SetDirty( existing );
            wildPrefab = AssetDatabase.LoadAssetAtPath<GameObject>( spec.ExistingWildPrefab );
        } else {
            var wildResource = CopyScriptable<ResourceSO>( WildResourceDir + spec.WildName + ".asset",
                WildResourceDir + ( spec.WildIsTree ? "WildAppleTree" : "Carrot" ) + ".asset", so => {
                    Set( so, "ItemSoOnGather", produce );
                    Set( so, "BonusItem", seed );
                    Set( so, "BonusChance", spec.WildSeedChance );
                    Set( so, "ToolTipMessage", spec.Display );
                    Set( so, "NumOfItemsOnGather", spec.WildCount );
                } );
            wildPrefab = BuildWildPrefab( spec, meshes, material, wildResource, spec.WildMesh ? spec.Name + "_Wild" : null );
        }

        var visual = BuildCropVisual( spec, meshes, material );
        var crop = AssetDatabase.LoadAssetAtPath<CropSO>( CropDir + spec.Name + ".asset" );
        if( crop == null ) {
            crop = ScriptableObject.CreateInstance<CropSO>();
            AssetDatabase.CreateAsset( crop, CropDir + spec.Name + ".asset" );
        }
        crop.lifecycle = spec.Lifecycle;
        crop.plotType = spec.Plot;
        crop.seed = seed;
        crop.produce = produce;
        crop.seedDropChance = spec.SeedDrop;
        crop.stageMinutes = spec.StageMinutes;
        crop.maxYield = spec.MaxYield;
        crop.fruitRegrowMinutes = spec.FruitRegrow;
        crop.visualPrefab = visual;
        crop.harvest = AssetDatabase.LoadAssetAtPath<CropSO>( CropDir + spec.HarvestFrom + ".asset" ).harvest;
        crop.forageKind = spec.Forage;
        EditorUtility.SetDirty( crop );

        foreach( var ( biome, probability, spacing ) in spec.Spawns ) {
            AddSpawn( BiomeDir + biome + ".asset", wildPrefab, probability, spacing );
        }
        Debug.Log( $"CropBatch2Builder: {spec.Name} ok" );
    }

    static Dictionary<string, Mesh> LoadMeshes( string name ) {
        var meshes = AssetDatabase.LoadAllAssetsAtPath( ModelDir + name + ".fbx" ).OfType<Mesh>().ToDictionary( m => m.name, m => m );
        Debug.Log( $"CropBatch2Builder: {name}.fbx meshes: {string.Join( ", ", meshes.Keys )}" );
        return meshes;
    }

    // ---- prefabs --------------------------------------------------------------------------------------------------

    static GameObject BuildWildPrefab( Spec spec, Dictionary<string, Mesh> meshes, Material material, ResourceSO resource, string wildMeshName ) {
        var mesh = wildMeshName != null ? meshes[ wildMeshName ] : meshes[ StageMeshName( spec, spec.Stages - 1 ) ];
        var root = new GameObject( spec.WildName ) { layer = ObjectLayer, tag = "SaveableObject" };
        NewChild( root.transform, "Model", mesh, material );
        if( wildMeshName == null && spec.Fruits > 0 ) {
            for( int i = 0; i < spec.Fruits; i++ ) {
                NewChild( root.transform, "Fruit" + i, meshes[ spec.Name + "_Fruit" + i ], material );
            }
        }
        var bounds = mesh.bounds;
        if( spec.WildIsTree ) {
            var capsule = root.AddComponent<CapsuleCollider>();
            capsule.radius = 0.12f;
            capsule.height = 0.7f;
            capsule.center = new Vector3( 0f, 0.35f, 0f );
        } else {
            var box = root.AddComponent<BoxCollider>();
            var size = bounds.size;
            size.x = Mathf.Max( size.x, 0.12f );
            size.z = Mathf.Max( size.z, 0.12f );
            size.y = Mathf.Max( size.y, 0.1f );
            box.size = size;
            box.center = bounds.center;
        }
        var res = root.AddComponent<Hearthglade.Gameplay.Resource.Resource>();
        var so = new SerializedObject( res );
        so.FindProperty( "resourceSO" ).objectReferenceValue = resource;
        so.ApplyModifiedPropertiesWithoutUndo();

        var prefab = PrefabUtility.SaveAsPrefabAsset( root, PrefabDir + spec.WildName + ".prefab" );
        UnityEngine.Object.DestroyImmediate( root );
        return prefab;
    }

    static string StageMeshName( Spec spec, int stage ) {
        return spec.Plot == PlotType.Orchard ? spec.Name + "_Tree" : spec.Name + "_Stage" + stage;
    }

    static GameObject BuildCropVisual( Spec spec, Dictionary<string, Mesh> meshes, Material material ) {
        var root = new GameObject( spec.Name + "Crop" ) { layer = ObjectLayer };
        bool orchard = spec.Plot == PlotType.Orchard;
        var origin = orchard ? new Vector3( CellHalf, OrchardY, CellHalf ) : spec.Plot == PlotType.Trellis ? Vector3.zero : new Vector3( 0f, BedY, 0f );
        var stageScales = orchard ? new[] { 0.257f, 0.57f, 1f } : new[] { 1f, 1f, 1f, 1f };

        var stages = new List<GameObject>();
        for( int i = 0; i < spec.Stages; i++ ) {
            var go = NewChild( root.transform, "Stage" + i, meshes[ StageMeshName( spec, i ) ], material );
            go.transform.localPosition = origin;
            go.transform.localScale = Vector3.one * stageScales[ i ];
            go.SetActive( i == spec.Stages - 1 );
            stages.Add( go );
        }
        var fruits = new List<GameObject>();
        for( int i = 0; i < spec.Fruits; i++ ) {
            var go = NewChild( root.transform, "Fruit" + i, meshes[ spec.Name + "_Fruit" + i ], material );
            go.transform.localPosition = origin;
            fruits.Add( go );
        }
        var visual = root.AddComponent<CropVisual>();
        var so = new SerializedObject( visual );
        FillArray( so.FindProperty( "stages" ), stages );
        FillArray( so.FindProperty( "fruits" ), fruits );
        so.ApplyModifiedPropertiesWithoutUndo();

        var prefab = PrefabUtility.SaveAsPrefabAsset( root, PrefabDir + spec.Name + "Crop.prefab" );
        UnityEngine.Object.DestroyImmediate( root );
        return prefab;
    }

    static GameObject NewChild( Transform parent, string name, Mesh mesh, Material material ) {
        var go = new GameObject( name ) { layer = ObjectLayer };
        go.transform.SetParent( parent, false );
        go.AddComponent<MeshFilter>().sharedMesh = mesh;
        go.AddComponent<MeshRenderer>().sharedMaterial = material;
        return go;
    }

    static void FillArray( SerializedProperty array, List<GameObject> objects ) {
        array.arraySize = objects.Count;
        for( int i = 0; i < objects.Count; i++ ) {
            array.GetArrayElementAtIndex( i ).objectReferenceValue = objects[ i ];
        }
    }

    static T CopyScriptable<T>( string path, string templatePath, Action<SerializedObject> edit ) where T : ScriptableObject {
        if( AssetDatabase.LoadAssetAtPath<T>( path ) == null ) {
            if( !AssetDatabase.CopyAsset( templatePath, path ) ) {
                throw new Exception( $"cannot copy {templatePath} to {path}" );
            }
        }
        var asset = AssetDatabase.LoadAssetAtPath<T>( path );
        var so = new SerializedObject( asset );
        edit( so );
        so.ApplyModifiedPropertiesWithoutUndo();
        EditorUtility.SetDirty( asset );
        return asset;
    }

    static void Set( SerializedObject so, string field, object value, bool optional = false ) {
        var p = so.FindProperty( field );
        if( p == null ) {
            if( optional ) {
                return;
            }
            throw new Exception( $"{so.targetObject.name}: no field {field}" );
        }
        switch( value ) {
            case string s: p.stringValue = s; break;
            case int i when p.propertyType == SerializedPropertyType.Enum: p.enumValueFlag = i; break;
            case int i: p.intValue = i; break;
            case float f: p.floatValue = f; break;
            case UnityEngine.Object o: p.objectReferenceValue = o; break;
            default: throw new Exception( $"{field}: unsupported {value.GetType()}" );
        }
    }

    static void SetIcon( SerializedObject so, string pngPath ) {
        var sprite = AssetDatabase.LoadAssetAtPath<Sprite>( pngPath );
        if( sprite == null ) {
            throw new Exception( "no sprite at " + pngPath );
        }
        so.FindProperty( "icon" ).objectReferenceValue = sprite;
    }

    static void AddSpawn( string biomePath, GameObject prefab, float probability, float spacing ) {
        var biome = AssetDatabase.LoadAssetAtPath<ScriptableObject>( biomePath );
        var so = new SerializedObject( biome );
        var list = so.FindProperty( "Resources" );
        for( int i = 0; i < list.arraySize; i++ ) {
            if( list.GetArrayElementAtIndex( i ).FindPropertyRelative( "resourcePrefab" ).objectReferenceValue == prefab ) {
                return;
            }
        }
        int index = list.arraySize;
        list.arraySize = index + 1;
        var e = list.GetArrayElementAtIndex( index );
        e.FindPropertyRelative( "resourcePrefab" ).objectReferenceValue = prefab;
        e.FindPropertyRelative( "howManyResourcesSpawn" ).intValue = 1;
        e.FindPropertyRelative( "spawnProbability" ).floatValue = probability;
        e.FindPropertyRelative( "spawnRange" ).floatValue = 0.3f;
        e.FindPropertyRelative( "minRotation" ).floatValue = 0f;
        e.FindPropertyRelative( "maxRotation" ).floatValue = 360f;
        e.FindPropertyRelative( "densityFrom" ).floatValue = 0f;
        e.FindPropertyRelative( "densityTo" ).floatValue = 0f;
        e.FindPropertyRelative( "water" ).enumValueIndex = 0;
        e.FindPropertyRelative( "minStartDistance" ).floatValue = 0f;
        e.FindPropertyRelative( "startDistanceRamp" ).floatValue = 0f;
        e.FindPropertyRelative( "minSpacing" ).floatValue = spacing;
        e.FindPropertyRelative( "clusterSpacing" ).intValue = 0;
        e.FindPropertyRelative( "clusterMin" ).intValue = 3;
        e.FindPropertyRelative( "clusterMax" ).intValue = 6;
        e.FindPropertyRelative( "clusterRadius" ).floatValue = 1.5f;
        so.ApplyModifiedPropertiesWithoutUndo();
        EditorUtility.SetDirty( biome );
    }

    static void RenderCrop( string cropName, PlotType plotType, int stages, string fileName = null ) {
        string plotPath = plotType == PlotType.Bed ? "Assets/_Prefabs/GardenPatch/GardenPatch.prefab"
            : plotType == PlotType.Orchard ? PrefabDir + "OrchardPlot.prefab" : PrefabDir + "Trellis.prefab";
        string cropPath = cropName == "Wheat" ? "Assets/_Prefabs/Environment/Crops/WheatCrop.prefab" : PrefabDir + cropName + "Crop.prefab";
        var plot = AssetDatabase.LoadAssetAtPath<GameObject>( plotPath );
        var crop = AssetDatabase.LoadAssetAtPath<GameObject>( cropPath );
        var root = new GameObject( "PreviewRoot" );
        float step = plotType == PlotType.Orchard ? 0.9f : 0.5f;
        for( int i = 0; i < stages; i++ ) {
            var pos = new Vector3( i * step, 0f, 0f );
            var p = ( GameObject ) PrefabUtility.InstantiatePrefab( plot, root.transform );
            p.transform.position = pos;
            foreach( var mb in p.GetComponentsInChildren<MonoBehaviour>() ) { mb.enabled = false; }
            var c = ( GameObject ) PrefabUtility.InstantiatePrefab( crop, root.transform );
            c.transform.position = pos;
            var so = new SerializedObject( c.GetComponent<CropVisual>() );
            var stageArr = so.FindProperty( "stages" );
            for( int s = 0; s < stageArr.arraySize; s++ ) {
                ( ( GameObject ) stageArr.GetArrayElementAtIndex( s ).objectReferenceValue ).SetActive( s == i );
            }
            var fruitArr = so.FindProperty( "fruits" );
            for( int f = 0; f < fruitArr.arraySize; f++ ) {
                ( ( GameObject ) fruitArr.GetArrayElementAtIndex( f ).objectReferenceValue ).SetActive( i == stages - 1 );
            }
        }
        var ground = GameObject.CreatePrimitive( PrimitiveType.Cube );
        ground.transform.SetParent( root.transform );
        ground.transform.position = new Vector3( ( stages - 1 ) * step * 0.5f, -0.05f, 0f );
        ground.transform.localScale = new Vector3( stages * step + 0.6f, 0.1f, 1.2f );
        var groundMat = new Material( Shader.Find( "Universal Render Pipeline/Lit" ) ?? Shader.Find( "Standard" ) );
        groundMat.color = new Color( 0.45f, 0.6f, 0.3f );
        ground.GetComponent<MeshRenderer>().sharedMaterial = groundMat;

        var lightGo = new GameObject( "Light" );
        var light = lightGo.AddComponent<Light>();
        light.type = LightType.Directional;
        light.intensity = 1.2f;
        lightGo.transform.rotation = Quaternion.Euler( 50f, -30f, 0f );

        var camGo = new GameObject( "Cam" );
        var cam = camGo.AddComponent<Camera>();
        cam.clearFlags = CameraClearFlags.SolidColor;
        cam.backgroundColor = new Color( 0.72f, 0.82f, 0.86f );
        cam.fieldOfView = 30f;
        float cx = ( stages - 1 ) * step * 0.5f;
        float height = plotType == PlotType.Orchard ? 0.35f : 0.18f;
        float dist = ( stages * step + 0.4f ) * 1.15f;
        camGo.transform.position = new Vector3( cx, height + dist * 0.45f, -dist * 0.8f );
        camGo.transform.LookAt( new Vector3( cx, height * 0.7f, 0.1f ) );
        var rt = new RenderTexture( 1400, 560, 24 );
        cam.targetTexture = rt;
        cam.Render();
        RenderTexture.active = rt;
        var tex = new Texture2D( rt.width, rt.height, TextureFormat.RGB24, false );
        tex.ReadPixels( new Rect( 0, 0, rt.width, rt.height ), 0, 0 );
        tex.Apply();
        File.WriteAllBytes( PreviewDir + "unity_" + ( fileName ?? cropName ) + ".png", tex.EncodeToPNG() );
        RenderTexture.active = null;
        UnityEngine.Object.DestroyImmediate( root );
        UnityEngine.Object.DestroyImmediate( lightGo );
        UnityEngine.Object.DestroyImmediate( camGo );
        UnityEngine.Object.DestroyImmediate( rt );
    }
